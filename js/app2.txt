// User jab kisi level (Beginner/Inter/Adv) par click kare
function selectLevel(levelName) {
    document.getElementById('levels-view').classList.add('hidden');
    document.getElementById('lessons-list-view').classList.remove('hidden');
    document.getElementById('current-level-title').textContent = `${levelName} Lessons`;

    const filtered = allLessons.filter(l => l.level === levelName);
    const container = document.getElementById('lessons-list');

    if (!filtered || filtered.length === 0) {
        container.innerHTML = `<p>No lessons available for ${levelName} yet.</p>`;
        return;
    }

    container.innerHTML = filtered.map(lesson => `
        <div class="lesson-card" style="display: flex; justify-content: space-between; align-items: center; padding: 1rem; border: 1px solid var(--border-color); border-radius: 8px; margin-bottom: 0.8rem;">
            <div>
                <h4 style="margin: 0;">${lesson.title}</h4>
                <p style="margin: 0.3rem 0 0 0; font-size: 0.85rem; color: var(--text-secondary);">${lesson.description || ''}</p>
            </div>
            <button class="btn btn-primary" onclick="startLesson(${lesson.id})">
                <i class="fa-solid fa-play"></i> Start
            </button>
        </div>
    `).join('');
}

// AI Backend ko request bhejne wala main function
async function handleSendMessage() {
    if (!chatInput) return;
    
    const userText = chatInput.value.trim();
    if (!userText) return;

    // 1. User ka message UI par dikhain
    appendMessage(userText, 'user');
    chatInput.value = '';

    // 2. Typing indicator (Bot is typing...)
    const typingIndicator = appendMessage('AI Buddy is thinking...', 'bot');

    try {
        // 3. Backend Express Server (/api/chat) ko call karein
        const response = await fetch('/api/chat', {
            method: 'POST',
            headers: { 
                'Content-Type': 'application/json' 
            },
            body: JSON.stringify({
                message: userText,
                targetLang: state.targetLang || 'Hindi',
                scenario: state.currentScenario || 'General'
            })
        });

        const data = await response.json();

        // Typing indicator hatain aur real AI response dikhain
        if (typingIndicator) typingIndicator.remove();

        if (data.reply) {
            appendMessage(data.reply, 'bot');
        } else {
            appendMessage("Sorry, I didn't get a response.", 'bot');
        }
    } catch (err) {
        if (typingIndicator) typingIndicator.remove();
        console.error("Chat API Error:", err);
        appendMessage("Network error. Please check backend server.", 'bot');
    }
}

// --- Interactive Lesson Quiz Logic ---
let currentLessonQuestions = [];
let currentQIndex = 0;
let lessonScore = 0;

// Lesson start karne ka handler
function startLesson(lessonId) {
    const lesson = allLessons.find(l => l.id === lessonId);
    if (!lesson || !lesson.questions || lesson.questions.length === 0) {
        alert("Is lesson me questions nahi hain.");
        return;
    }

    currentLessonQuestions = lesson.questions;
    currentQIndex = 0;
    lessonScore = 0;

    document.getElementById('lessons-list-view').classList.add('hidden');
    document.getElementById('lesson-quiz-view').classList.remove('hidden');

    renderLessonQuestion();
}

// Question Screen Render logic
function renderLessonQuestion() {
    const q = currentLessonQuestions[currentQIndex];
    const targetLang = state.targetLang || 'mr'; // Selected language (Marathi/Hindi)
    
    // Question Header Dynamic Set
    const questionText = q.question[targetLang] || q.question['mr'] || q.question['hi'];
    document.getElementById('lesson-quiz-question').textContent = questionText;
    document.getElementById('lesson-quiz-progress').textContent = `Question ${currentQIndex + 1} of ${currentLessonQuestions.length}`;
    
    document.getElementById('feedback-message').textContent = '';
    document.getElementById('next-lesson-q-btn').classList.add('hidden');

    const optionsGrid = document.getElementById('lesson-quiz-options');
    optionsGrid.innerHTML = '';

    // Options Button Generate
    q.options.forEach((optionText, index) => {
        const btn = document.createElement('button');
        btn.className = 'option-btn';
        btn.textContent = optionText;
        btn.onclick = () => checkLessonAnswer(index, q.correctIndex, btn);
        optionsGrid.appendChild(btn);
    });
}

// Answer Check logic
function checkLessonAnswer(selectedIndex, correctIndex, selectedBtn) {
    const allBtns = document.querySelectorAll('#lesson-quiz-options .option-btn');
    allBtns.forEach(btn => btn.disabled = true); // Block multiple clicks

    const feedbackEl = document.getElementById('feedback-message');

    if (selectedIndex === correctIndex) {
        selectedBtn.classList.add('correct');
        feedbackEl.textContent = "🎉 Sahi Jawab! (Correct)";
        feedbackEl.style.color = "var(--success-color)";
        if (typeof audioEngine !== 'undefined') audioEngine.playCorrect();
        lessonScore += 10;
        if (typeof state !== 'undefined' && state.addXp) state.addXp(10);
        if (typeof updateStatsDisplay === 'function') updateStatsDisplay();
    } else {
        selectedBtn.classList.add('incorrect');
        allBtns[correctIndex].classList.add('correct'); // Sahi answer highlight karein
        feedbackEl.textContent = "❌ Galat Jawab! Sahi option highlighted hai.";
        feedbackEl.style.color = "var(--danger-color)";
        if (typeof audioEngine !== 'undefined') audioEngine.playIncorrect();
    }

    document.getElementById('next-lesson-q-btn').classList.remove('hidden');
}

// Next Question Handler
function nextLessonQuestion() {
    currentQIndex++;
    if (currentQIndex < currentLessonQuestions.length) {
        renderLessonQuestion();
    } else {
        // Lesson Complete Screen
        document.getElementById('lesson-quiz-question').textContent = `🏆 Lesson Complete! Total Score: ${lessonScore} XP`;
        document.getElementById('lesson-quiz-options').innerHTML = '';
        document.getElementById('feedback-message').textContent = 'Well Done!';
        document.getElementById('next-lesson-q-btn').classList.add('hidden');
    }
}

    document.getElementById('lessons-list-view').classList.remove('hidden');
}
